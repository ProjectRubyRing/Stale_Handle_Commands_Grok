# java.io.IOException: Stale file handle 調査用コマンド
# 対象: ECS Exec で入った Fargate コンテナ（UBI 9.8）の /mnt/logs
# 日付の前提: 2026-10-14。RHEL 9.8 は 2026-05-19 GA、カーネル 5.14.0-687.5.1.el9_8。
#
# 貼り方
# 1. ECS Exec のコマンドは /bin/bash にする。プロンプトが bash であることを確認する。
# 2. このファイルの先頭から、単独行の ESTALE_PROBE_END までをまとめて貼る。
# 3. 一括調査は読み取り専用。umount も rm も kill もしない。結果は標準出力に出る。
# 4. 画面の最後にある SUMMARY と READ 行を、セッションの外へコピーして残す。
# 5. ファイル全体を貼っても、一括調査のあとの個別ブロックは実行されない。
#    個別ブロックは、必要な番号の set +H から ESTALE_BLOCK_番号 までを選んで貼る。
# 6. 手元の PowerShell では実行しない。中身はコンテナの bash 用である。
# 7. パスを変えるときだけ、貼る前に LOGROOT=/実際のパス を export する。既定は /mnt/logs。
# 8. 行頭の # は説明であり、bash は実行しない。

mark_timeout() {
  if [[ "$1" -eq 124 ]]; then
    HINT_TIMEOUT=$((HINT_TIMEOUT + 1))
    echo "NOTE: exit 124 means the time limit fired. A process blocked in NFS D state can ignore SIGKILL until the server answers. Open another ECS Exec session if this shell stops printing."
  fi
}

mark_estale_text() {
  case "$1" in
    *"Stale file handle"*|*"stale file handle"*|*"Stale NFS file handle"*|*"古いファイルハンドルです"*|*"実効性のないNFSファイルハンドルです"*)
      HINT_SAW_ESTALE=1
      ;;
  esac
}

run_cmd() {
  local sec=$1
  shift
  local out rc
  printf '+'
  printf ' %q' "$@"
  printf '\n'
  if have timeout; then
    out=$(timeout "$sec" "$@" 2>&1)
    rc=$?
  else
    out=$("$@" 2>&1)
    rc=$?
  fi
  printf '%s\n' "$out" | head -n 220
  echo "exit=${rc}"
  mark_timeout "$rc"
  mark_estale_text "$out"
  LAST_OUT=$out
  LAST_RC=$rc
}

mask_word() {
  local word=$1
  local lower
  lower=$(printf '%s' "$word" | tr '[:upper:]' '[:lower:]')
  case "$lower" in
    *password=*|*passwd=*|*secret=*|*token=*|*credential=*|*accesskey=*|*secretkey=*|*access_key=*|*session_token=*|*private_key=*)
      printf '%s\n' "${word%%=*}=***MASKED***"
      ;;
    *)
      printf '%s\n' "$word"
      ;;
  esac
}

mask_line() {
  local line=$1
  local word out=""
  set -f
  for word in $line; do
    out="${out} $(mask_word "$word")"
  done
  set +f
  printf '%s\n' "${out# }"
}

}

decode_ipv4() {
  local spec=$1
  local hex=${spec%:*}
  local port=${spec#*:}
  local b1=${hex:6:2}
  local b2=${hex:4:2}
  local b3=${hex:2:2}
  local b4=${hex:0:2}
  printf '%d.%d.%d.%d:%d\n' "0x${b1}" "0x${b2}" "0x${b3}" "0x${b4}" "0x${port}"
}

state_name() {
  case "$1" in
    01) printf '%s\n' ESTABLISHED ;;
    02) printf '%s\n' SYN_SENT ;;
    03) printf '%s\n' SYN_RECV ;;
    04) printf '%s\n' FIN_WAIT1 ;;
    05) printf '%s\n' FIN_WAIT2 ;;
    06) printf '%s\n' TIME_WAIT ;;
    07) printf '%s\n' CLOSE ;;
    08) printf '%s\n' CLOSE_WAIT ;;
    09) printf '%s\n' LAST_ACK ;;
    0A|0a) printf '%s\n' LISTEN ;;
    0B|0b) printf '%s\n' CLOSING ;;
    0C|0c) printf '%s\n' NEW_SYN_RECV ;;
    *) printf '%s\n' "UNKNOWN_${1}" ;;
  esac
}

decode_flags() {
  local oct=$1
  local n=$((8#$oct))
  local acc=$((n & 3))
  local out=""
  if [[ $acc -eq 0 ]]; then out="O_RDONLY"; fi
  if [[ $acc -eq 1 ]]; then out="O_WRONLY"; fi
  if [[ $acc -eq 2 ]]; then out="O_RDWR"; fi
  local bit name masked
  local bits=(64 128 512 1024 2048 4096 16384 32768 65536 131072 262144 524288 1048576 2097152)
  local names=(O_CREAT O_EXCL O_TRUNC O_APPEND O_NONBLOCK O_DSYNC O_DIRECT O_LARGEFILE O_DIRECTORY O_NOFOLLOW O_NOATIME O_CLOEXEC O_SYNC_BIT O_PATH)
  local i=0
  local count=${#bits[@]}
  while [[ $i -lt $count ]]; do
    bit=${bits[$i]}
    name=${names[$i]}
    masked=$((n & bit))
    if [[ $masked -eq $bit ]]; then
      out="${out} ${name}"
    fi
    i=$((i + 1))
  done
  printf '%s\n' "$out"
}

label_numbers() {
  local kind=$1
  shift
  local -a names=("$@")
  local n=${#LAST_NUMS[@]}
  local i name
  echo "${kind}_count=${n} known_names=${#names[@]}"
  i=0
  while [[ $i -lt $n ]]; do
    if [[ $i -lt ${#names[@]} ]]; then
      name=${names[$i]}
    else
      name="extra_$((i + 1))"
    fi
    printf '  %s=%s\n' "$name" "${LAST_NUMS[$i]}"
    if [[ "$name" == "sillyrename" ]]; then
      if [[ "${LAST_NUMS[$i]}" == "0" ]]; then
        :
      else
        HINT_SILLY=1
        note "HINT sillyrename_counter=${LAST_NUMS[$i]}"
      fi
    fi
    i=$((i + 1))
  done
}

split_nums() {
  LAST_NUMS=()
  local rest=$1
  read -r -a LAST_NUMS <<< "$rest"
}

label_op() {
  local trim=$1
  local name=${trim%%:*}
  local rest=${trim#*:}
  split_nums "$rest"
  local n=${#LAST_NUMS[@]}
  if [[ $n -lt 3 ]]; then
    echo "OP_RAW ${trim}"
    return
  fi
  local ops=${LAST_NUMS[0]}
  local trans=${LAST_NUMS[1]}
  local timeouts=${LAST_NUMS[2]}
  local retrans=$((trans - ops))
  local errors="absent"
  if [[ $n -ge 9 ]]; then
    errors=${LAST_NUMS[8]}
  fi
  if [[ "$ops" == "0" ]]; then
    return
  fi
  echo "OP ${name} ops=${ops} trans=${trans} retrans=${retrans} timeouts=${timeouts} errors=${errors} fields=${n}"
  case "$name" in
    READ|WRITE|COMMIT|OPEN|CLOSE|CREATE|REMOVE|RENAME|LINK|SETATTR|GETATTR|ACCESS|LOOKUP)
      echo "OP_DETAIL ${name} raw=${rest}"
      ;;
  esac
}

selftest() {
  local fail=0
  local got
  got=$(decode_ipv4 "0100007F:0801")
  if [[ "$got" == "127.0.0.1:2049" ]]; then
    echo "SELFTEST ipv4 ok"
  else
    echo "SELFTEST ipv4 FAIL got=${got}"
    fail=1
  fi
  got=$(decode_flags "02100002")
  if [[ "$got" == *O_RDWR* && "$got" == *O_LARGEFILE* && "$got" == *O_CLOEXEC* ]]; then
    if [[ "$got" == *O_APPEND* ]]; then
      echo "SELFTEST flags FAIL append unexpectedly in ${got}"
      fail=1
    else
      echo "SELFTEST flags ok ${got}"
    fi
  else
    echo "SELFTEST flags FAIL got=${got}"
    fail=1
  fi
  if [[ "${EVENT_NAMES[21]}" == "sillyrename" ]]; then
    echo "SELFTEST events ok"
  else
    echo "SELFTEST events FAIL ${EVENT_NAMES[21]}"
    fail=1
  fi
  got=$(state_name "01")
  if [[ "$got" == "ESTABLISHED" ]]; then
    echo "SELFTEST state ok"
  else
    echo "SELFTEST state FAIL"
    fail=1
  fi
  return "$fail"
}

declare -a LOG_CANDIDATES=()
walk() {
  local dir=$1
  local depth=$2
  local path base line mtime size ino ftype dev age now
  if [[ $depth -gt 4 ]]; then
    return
  fi
  for path in "$dir"/*; do
    if [[ $WALK_COUNT -ge $WALK_MAX ]]; then
      echo "WALK_TRUNCATED max=${WALK_MAX}"
      return
    fi
    WALK_COUNT=$((WALK_COUNT + 1))
    if [[ -L "$path" ]]; then
      echo "SYMLINK ${path} -> $(readlink -- "$path" 2>/dev/null)"
      continue
    fi
    line=$(stat -c '%F|%i|%s|%Y|%d|%n' -- "$path" 2>&1)
    rc=$?
    if [[ $rc -ne 0 ]]; then
      echo "STAT_FAIL ${path} :: ${line}"
      mark_estale_text "$line"
      case "$line" in
        *"Stale file handle"*) HINT_DIR_ESTALE=1 ;;
      esac
      continue
    fi
    ftype=${line%%|*}
    rest=${line#*|}
    ino=${rest%%|*}
    rest=${rest#*|}
    size=${rest%%|*}
    rest=${rest#*|}
    mtime=${rest%%|*}
    rest=${rest#*|}
    dev=${rest%%|*}
    printf 'ENTRY type=%s inode=%s size=%s mtime=%s dev=%s path=%s\n' "$ftype" "$ino" "$size" "$mtime" "$dev" "$path"
    base=${path##*/}
    case "$base" in
      .nfs*)
        HINT_DOTNFS=$((HINT_DOTNFS + 1))
        echo "DOTNFS ${path}"
        ;;
      *.lck)
        HINT_LCK=$((HINT_LCK + 1))
        echo "LOCKFILE ${path}"
        ;;
    esac
    if [[ "$size" == "0" ]]; then
      echo "ZERO_BYTE ${path}"
    fi
    case "$base" in
      *.log|*.out|*.err|*.txt|*.log.*|*.out.*)
        case "$base" in
          *.gz|*.zip|*.xz) ;;
          *) LOG_CANDIDATES+=("${mtime}|${path}") ;;
        esac
        ;;
    esac
    case "$ftype" in
      directory*)
        if [[ -n "$ROOT_DEV" ]]; then
          if [[ "$dev" == "$ROOT_DEV" ]]; then
            walk "$path" $((depth + 1))
          else
            echo "SKIP_OTHER_DEVICE ${path}"
          fi
        else
          walk "$path" $((depth + 1))
        fi
        ;;
    esac
  done
}

      for fdpath in "$d"/fd/*; do
        target=$(readlink "$fdpath" 2>/dev/null)
        case "$target" in
          "${LOGROOT}"|"${LOGROOT}"/*|*" (deleted)")
            case "$target" in
              "${LOGROOT}"|"${LOGROOT}"/*|*"${LOGROOT}"*) holds=1 ;;
            esac
            ;;
        esac
      done
    fi
    if [[ $is_java -eq 0 && $holds -eq 0 ]]; then
      continue
    fi
    if [[ $is_java -eq 1 ]]; then
      HINT_JAVA=$((HINT_JAVA + 1))
    fi
    section "10 pid ${pid} comm=${comm} java=${is_java} holds_log_or_deleted=${holds}"
    echo "exe=${exe}"
    echo "cmdline_masked=$(mask_line "$cmdline")"
    echo "-- status --"
    while IFS= read -r line; do
      case "$line" in
        State:*|Uid:*|Gid:*|NStgid:*|Threads:*|Pid:*) printf '%s\n' "$line" ;;
      esac
    done < "$d/status"
    echo "-- limits --"
    if [[ -r "$d/limits" ]]; then
      while IFS= read -r line; do
        case "$line" in
          "Max open files"*|"Max file locks"*) printf '%s\n' "$line" ;;
        esac
      done < "$d/limits"
    fi
    fds=("$d"/fd/*)
    echo "fd_count=${#fds[@]}"
    echo "-- environ keys (values only for an allow list, secrets masked) --"
    if [[ -r "$d/environ" ]]; then
      while IFS= read -r -d '' ev; do
        key=${ev%%=*}
        val=${ev#*=}
        if value_allowed "$key"; then
          printf '%s=%s\n' "$key" "$(mask_line "$val")"
        else
          printf '%s=SET\n' "$key"
        fi
      done < "$d/environ"
    else
      echo "environ not readable"
    fi
    echo "-- fds pointing at LOGROOT or deleted --"
    for fdpath in "$d"/fd/*; do
      base=${fdpath##*/}
      target=$(readlink "$fdpath" 2>&1)
      rc=$?
      if [[ $rc -ne 0 ]]; then
        echo "READLINK_FAIL fd=${base} :: ${target}"
        continue
      fi
      related=0
      deleted=0
      case "$target" in
        *" (deleted)") deleted=1 ;;
      esac
      case "$target" in
        "${LOGROOT}"|"${LOGROOT}"/*) related=1 ;;
        *" (deleted)")
          bare=${target% (deleted)}
          case "$bare" in
            "${LOGROOT}"|"${LOGROOT}"/*) related=1 ;;
          esac
          ;;
      esac
      if [[ $related -eq 0 ]]; then
        continue
      fi
      echo "FD ${base} -> ${target}"
      fdstat=$(stat -c 'ftype=%F inode=%i device=%d size=%s links=%h' -- "$fdpath" 2>&1)
      echo "FD_STAT ${fdstat}"
      mark_estale_text "$fdstat"
      bare=$target
      if [[ $deleted -eq 1 ]]; then
        bare=${target% (deleted)}
        HINT_DELETED=$((HINT_DELETED + 1))
        echo "DELETED_MARKER ${bare}"
      fi
      pathstat=$(stat -c 'ftype=%F inode=%i device=%d size=%s links=%h' -- "$bare" 2>&1)
      echo "PATH_STAT ${pathstat}"
      mark_estale_text "$pathstat"
      fd_ino=""
      path_ino=""
      case "$fdstat" in
        *"inode="*)
          tmp=${fdstat#*inode=}
          fd_ino=${tmp%% *}
          ;;
      esac
      case "$pathstat" in
        *"inode="*)
          tmp=${pathstat#*inode=}
          path_ino=${tmp%% *}
          ;;
      esac
      if [[ -n "$fd_ino" && -n "$path_ino" ]]; then
        if [[ "$fd_ino" == "$path_ino" ]]; then
          echo "INODE_MATCH ${fd_ino}"
        else
          echo "INODE_MISMATCH fd=${fd_ino} path=${path_ino} file=${bare}"
          HINT_MISMATCH=$((HINT_MISMATCH + 1))
        fi
      fi
      if [[ -r "$d/fdinfo/$base" ]]; then
        echo "-- fdinfo ${base} --"
        flags_oct=""
        while IFS= read -r fline; do
          printf '%s\n' "$fline"
          case "$fline" in
            flags:*)
              flags_oct=${fline#flags:}
              flags_oct=${flags_oct//[[:space:]]/}
              ;;
          esac
        done < "$d/fdinfo/$base"
        case "$flags_oct" in
          [0-7]*)
            echo "FLAGS_DECODE $(decode_flags "$flags_oct")"
            ;;
          "")
            ;;
          *)
            echo "FLAGS_UNPARSED ${flags_oct}"
            ;;
        esac
      fi
      case "$fdstat" in
        *"regular file"*|*"regular empty file"*)
          if [[ ${TAILED:-0} -lt 3 ]]; then
            echo "-- read-only tail of this regular-file fd --"
            if have timeout; then
              tout=$(timeout 10 tail -n 12 -- "$fdpath" 2>&1 | tr -d '\000' | head -c 1200)
            else
              tout=$(tail -n 12 -- "$fdpath" 2>&1 | tr -d '\000' | head -c 1200)
            fi
            printf '%s\n' "$tout"
            echo
            mark_estale_text "$tout"
            TAILED=$((${TAILED:-0} + 1))
          fi
          ;;
      esac
    done
    TAILED=0
  done
fi

# C24 mountstats から /mnt/logs の節を抜く
# 危険度: 読み取り専用。ファイルが無い構成がある
# 実行場所: コンテナの bash
set +H
LOGROOT="${LOGROOT:-/mnt/logs}" bash <<'ESTALE_BLOCK_C24'
show=0
while IFS= read -r line
do
  case "$line" in
    device\ *)
      show=0
      case "$line" in
        *" mounted on ${LOGROOT} "*) show=1 ;;
      esac
      ;;
  esac
  if [[ "$show" == "1" ]]
  then
    printf '%s\n' "$line"
  fi
done < /proc/self/mountstats
ESTALE_BLOCK_C24

# C26 WRITE と REMOVE と RENAME の回数を見る
# 危険度: 読み取り専用
# 実行場所: コンテナの bash
set +H
LOGROOT="${LOGROOT:-/mnt/logs}" bash <<'ESTALE_BLOCK_C26'
echo "列は ops trans timeouts bytes_sent bytes_recv queue_ms rtt_ms execute_ms errors"
echo "errors は 9 列目。RPC iostats 1.1 以降。再送の目安は trans から ops を引く。"
while IFS= read -r line
do
  trim=$line
  while [[ "$trim" == [[:space:]]* ]]
  do
    trim=${trim#?}
  done
  case "$trim" in
    WRITE:*|READ:*|COMMIT:*|OPEN:*|CLOSE:*|CREATE:*|REMOVE:*|RENAME:*|SETATTR:*|GETATTR:*)
      printf '%s\n' "$trim"
      ;;
  esac
done < /proc/self/mountstats
ESTALE_BLOCK_C26
